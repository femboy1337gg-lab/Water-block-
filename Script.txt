local function fireBlock()
    if not Remote then return end
    if tick() - lastBlock < BLOCK_COOLDOWN then return end
    lastBlock = tick()
    pcall(function()
        Remote:FireServer("UseActorAbility", {"Block"})
    end)
    print("[AB] âœ“ Block fired")
end

local function firePunch()
    if not Remote then return end
    pcall(function()
        Remote:FireServer("UseActorAbility", {"Punch"})
    end)
end

local function getMyRoot()
    local char = LocalPlayer.Character
    return char and char:FindFirstChild("HumanoidRootPart")
end

local function getKillers()
    local folder = workspace:FindFirstChild("Players")
    return folder and folder:FindFirstChild("Killers")
end

local function isInRound()
    local char = LocalPlayer.Character
    if not char then return false end
    local survivors = workspace:FindFirstChild("Players") and workspace.Players:FindFirstChild("Survivors")
    if not survivors then return false end
    return char:IsDescendantOf(survivors)
end

-- ====================== INF STAMINA ======================
RunService.Heartbeat:Connect(function()
    if not Settings.InfStamina or not isInRound() then return end
    local char = LocalPlayer.Character
    if not char then return end

    local attr = char:GetAttribute("Stamina") or char:GetAttribute("stamina")
    if typeof(attr) == "number" and attr <= 1.5 then
        pcall(function()
            char:SetAttribute("Stamina", 100)
            char:SetAttribute("stamina", 100)
        end)
    end

    for _, v in ipairs(char:GetDescendants()) do
        if (v:IsA("NumberValue") or v:IsA("IntValue")) and v.Name:lower():find("stam") then
            if v.Value <= 1.5 then
                v.Value = 100
            end
        end
    end
end)

-- ====================== PUNCH AIM ======================
local function aimAtNearestKiller()
    if not Settings.PunchAim then return end
    local myRoot = getMyRoot()
    if not myRoot then return end

    local closest, closestDist = nil, math.huge
    local killers = getKillers()
    if killers then
        for _, killer in ipairs(killers:GetChildren()) do
            local kRoot = killer:FindFirstChild("HumanoidRootPart")
            if kRoot then
                local d = (myRoot.Position - kRoot.Position).Magnitude
                if d < closestDist then
                    closest = kRoot
                    closestDist = d
                end
            end
        end
    end

    if closest and closestDist < 45 then
        local look = (closest.Position - myRoot.Position) * Vector3.new(1, 0, 1)
        if look.Magnitude > 0.05 then
            myRoot.CFrame = CFrame.lookAt(myRoot.Position, myRoot.Position + look)
            local cam = workspace.CurrentCamera
            if cam then
                cam.CFrame = CFrame.new(cam.CFrame.Position, closest.Position)
            end
        end
    end
end

local function updateVisuals()
    local myRoot = getMyRoot()

-- ====================== AUTO BLOCK (restored working logic) ======================
local killerConns = {}

local function onM1Detected(killerRoot)
    if not Settings.AutoBlock then return end
    if not isInRound() then return end

    local myRoot = getMyRoot()
    if not myRoot then return end

    local dist = (myRoot.Position - killerRoot.Position).Magnitude
    if dist > Settings.Range then return end

    if Settings.FaceCheck then
        local dir = (myRoot.Position - killerRoot.Position).Unit
        local facing = killerRoot.CFrame.LookVector:Dot(dir)
        if facing < Settings.FacingDot then
            return
        end
    end

    print("[AB] M1 detected - blocking")
    task.delay(Settings.BlockDelay, function()
        if not Settings.AutoBlock then return end
        fireBlock()

        if Settings.AutoPunch then
            task.delay(0.28, function()
                if Settings.AutoBlock then
                    if Settings.PunchAim then aimAtNearestKiller() end
                    firePunch()
                end
            end)
        end

        if Settings.HitboxDrag then
            hdtTarget = killerRoot
            hdtActiveUntil = tick() + Settings.HDT_Duration
        end
    end)
end

local function hookKiller(killer)
    if killerConns[killer] then return end
    local root = killer:FindFirstChild("HumanoidRootPart") or killer:WaitForChild("HumanoidRootPart", 4)
    if not root then return end

    local conn = root.DescendantAdded:Connect(function(obj)
        if obj:IsA("Sound") and M1Sounds[obj.SoundId] then
            onM1Detected(root)
        end
    end)
    killerConns[killer] = conn
    print("[AB] Hooked killer:", killer.Name)
end

-- Global sound listener (backup)
workspace.DescendantAdded:Connect(function(obj)
    if obj:IsA("Sound") then
        obj.Played:Connect(function()
            if not M1Sounds[obj.SoundId] then return end
            if not Settings.AutoBlock or not isInRound() then return end

            local myRoot = getMyRoot()
            if not myRoot then return end

            local killers = getKillers()
            if not killers then return end

            for _, killer in ipairs(killers:GetChildren()) do
                local kRoot = killer:FindFirstChild("HumanoidRootPart")
                if kRoot then
                    local dist = (myRoot.Position - kRoot.Position).Magnitude
                    if dist <= Settings.Range + 3 then
                        onM1Detected(kRoot)
                        break
                    end
                end
            end
        end)
    end
end)

local function startAutoBlock(state)
    Settings.AutoBlock = state
    local killers = getKillers()

    if state then
        if killers then
            for _, killer in ipairs(killers:GetChildren()) do
                hookKiller(killer)
            end
            killers.ChildAdded:Connect(hookKiller)
        end
        print("[AB] Auto Block ENABLED")
    else
        for _, conn in pairs(killerConns) do
            conn:Disconnect()
        end
        table.clear(killerConns)
        hdtActiveUntil = 0
        hdtTarget = nil
        print("[AB] Auto Block DISABLED")
    end
end

-- HDT
RunService.Heartbeat:Connect(function()
    if not Settings.HitboxDrag then return end
    if tick() > hdtActiveUntil or not hdtTarget or not hdtTarget.Parent then
        hdtTarget = nil
        return
    end
    if not isInRound() then return end

    local myRoot = getMyRoot()
    if not myRoot then return end

    local dist = (hdtTarget.Position - myRoot.Position).Magnitude
    if dist < 4 then return end

    local dir = (hdtTarget.Position - myRoot.Position).Unit
    local speed = Settings.HitboxSpeed
    if dist > 10 then speed = speed + 5 end

    local shake = Settings.HitboxShake and Vector3.new(
        math.random(-10,10)/100, math.random(-5,5)/100, math.random(-10,10)/100
    ) or Vector3.zero

    myRoot.Velocity = dir * speed + shake
end)
